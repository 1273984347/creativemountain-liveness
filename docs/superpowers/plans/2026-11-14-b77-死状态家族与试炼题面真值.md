```ts
// B77 死状态家族的**唯一**读写出处（规格 B77 §4：两个家族、各自单一出处）。
//
// 分工：本文件管「写入侧」（applyStatus / grantImmunity）与「家族 B 结算侧」
//   （tickPlayerStatuses，Task 4 加入）；**力竭（家族 A）不在这里**——它留在
//   turn-runner 的 endRound（B76 r1.2 定稿，exhaustion.test.ts:196/246 钉着）。
//
// ⚠️ 日志 kind 一律 'status'，message 前缀是**跨任务契约**（模拟器按前缀扫计数器、
//   测试逐字匹配）：六个前缀见规格 §9 表，改一个字都要先改规格再改这里。
import type { NegativeStatus } from '../types/common.ts';
import type { State } from './state.ts';
import { applyEffect } from '../ops/registry.ts';
import { aliveTarget, fail, ok, withLog, withPlayer, type OpContext, type OpResult } from '../ops/registry.ts';

/** `flags` 里免疫窗的键前缀（immunity 寄居 flags，不迁容器——理由见规格 §3.5）。 */
export const IMMUNITY_FLAG_PREFIX = 'immunity_';

/** 统一日志 kind。 */
const KIND = 'status';

/** 读层数 / 回合数：旧存档缺字段一律 0（B75 R4 容错惯例）。 */
export function statusOf(state: State, seat: number, kind: NegativeStatus): number {
  return state.players[seat]?.statuses?.[kind] ?? 0;
}

/** 读免疫窗剩余回合。 */
export function immunityOf(state: State, seat: number, kind: NegativeStatus): number {
  return (state.players[seat]?.flags[IMMUNITY_FLAG_PREFIX + kind] as number | undefined) ?? 0;
}

/** 置「本手番授予过」标记（家族 B 递减豁免的依据，规格 §4 豁免条）。 */
function markGranted(state: State, seat: number): State {
  const p = state.players[seat];
  if (p === undefined) { return state; }
  return withPlayer(state, seat, { turn: { ...p.turn, statusGrantedThisTurn: true } });
}

/**
 * 给 `seat` 加 `amount` 层 / 回合的负面状态。
 *
 * ⚠️ **累加，不是设层**（规格 §3 通用形状）：`poison`/`slow` 加层＝当回合强度，
 *   `silence`/`skip_turn` 加回合＝持续时间；卡面「使其中毒 1 层」的「使其」是累加语气。
 * ⚠️ 免疫窗在**此拦截**：挡住的是整件「被授予」事件——封口连弃牌一起挡（规格 §3.3）。
 */
export function applyStatus(ctx: OpContext, seat: number, kind: NegativeStatus, amount: number): OpResult {
  if (amount <= 0) { return fail(ctx.state, 'amount 必须为正'); }
  const t = aliveTarget(ctx, seat);
  if (!t.ok) { return fail(ctx.state, t.note); }
  const state = ctx.state;
  if (immunityOf(state, t.seat, kind) > 0) {
    return ok(withLog(state, { ...ctx, actor: t.seat }, KIND, `被免疫：座位 ${String(t.seat)} 的 ${kind}`));
  }
  const p = state.players[t.seat]!;
  let next = withPlayer(state, t.seat, { statuses: { ...p.statuses, [kind]: (p.statuses?.[kind] ?? 0) + amount } });
  next = markGranted(next, t.seat);
  next = withLog(next, { ...ctx, actor: t.seat }, KIND, `授予：座位 ${String(t.seat)} 的 ${kind} +${String(amount)}`);
  // 封口＝弃牌 + 禁言，弃牌是**写入那一刻**的即时代价（规格 §3.3）。
  // 复用 discard_self：它是「弃 N 张」的唯一实现（弃首位、进弃牌堆、自带日志）。
  if (kind === 'silence' && (p.hand.length ?? 0) > 0) {
    const dropped = p.hand[0]!;
    const r = applyEffect({ state: next, actor: t.seat }, { op: 'discard_self', count: 1 });
    next = r.ok
      ? withLog(r.state, { ...ctx, actor: t.seat }, KIND, `封口弃牌：座位 ${String(t.seat)}（弃 ${dropped.name}）`)
      : next;
  }
  return ok(next);
}

/** 授予免疫窗（`immunity` 算子的落点）。 */
export function grantImmunity(ctx: OpContext, seat: number, status: NegativeStatus, amount: number): OpResult {
  if (amount <= 0) { return fail(ctx.state, 'amount 必须为正'); }
  const t = aliveTarget(ctx, seat);
  if (!t.ok) { return fail(ctx.state, t.note); }
  const p = ctx.state.players[t.seat]!;
  let next = withPlayer(ctx.state, t.seat, {
    flags: { ...p.flags, [IMMUNITY_FLAG_PREFIX + status]: immunityOf(ctx.state, t.seat, status) + amount },
  });
  next = markGranted(next, t.seat);
  return ok(withLog(next, { ...ctx, actor: t.seat }, KIND, `免疫 ${status} ${String(amount)} 回合：座位 ${String(t.seat)}`));
}
```

```ts
/** 家族 B 的四个衰减对象 + poison 的结算对象（顺序即规格 §4 的 ②–⑥）。 */
const TICK_ORDER = ['slow', 'silence', 'skip_turn'] as const;
const IMMUNITY_TICK = ['poison', 'slow', 'silence', 'skip_turn'] as const;

/** 把某键 −1（下限 0），返回新 state。 */
function decStatus(state: State, seat: number, kind: NegativeStatus): State {
  const p = state.players[seat]!;
  return withPlayer(state, seat, { statuses: { ...p.statuses, [kind]: Math.max(0, (p.statuses?.[kind] ?? 0) - 1) } });
}

/**
 * 家族 B 的**唯一**结算处（规格 B77 §4：② poison 扣血 → ③④⑤ slow/silence/skip 衰减 → ⑥ immunity 衰减）。
 *
 * ⚠️ 为什么锚在**该玩家手番末**而不是 round 末：每人每 round 恰好一个手番
 *   （round.ts:17-27 + isRoundEnd），而三个写入点全在该玩家自己 手番内 ⇒ round 末递减
 *   会在其下次行动**之前**把状态抹平 ⇒「停 1 回合」停不到、禁言永不生效、免疫窗恒 0。
 *   力竭（家族 A）不同：它是 round 级计数器，留在 endRound 不动（B76 r1.2 定稿）。
 * ⚠️ 死者整体跳过（I2 同族）：既不扣血也不衰减也不写日志。
 */
export function tickPlayerStatuses(state: State, seat: number): State {
  const p0 = state.players[seat];
  if (p0 === undefined || !p0.alive) { return state; }
  const granted = p0.turn.statusGrantedThisTurn === true;
  let next = state;

  // ② poison：先按层数扣血（clamp 0）⇒ 归零走 triggerExhaustion ⇒ 再 −1 层。
  const layers = statusOf(next, seat, 'poison');
  if (layers > 0) {
    const p = next.players[seat]!;
    const hp = Math.max(0, p.hp - layers);
    next = withPlayer(next, seat, { hp });
    next = withLog(next, { state: next, actor: seat }, 'status', `中毒扣血：座位 ${String(seat)}（-${String(layers)}）`);
    if (hp === 0) { next = triggerExhaustion(next, seat); }
    next = decStatus(next, seat, 'poison');
  }

  // ③④⑤ 与 ⑥：授予当回合整体豁免（不吃刚挂上的那一拍），段末自清标记。
  if (!granted) {
    for (const kind of TICK_ORDER) { if (statusOf(next, seat, kind) > 0) { next = decStatus(next, seat, kind); } }
    for (const kind of IMMUNITY_TICK) {
      const left = immunityOf(next, seat, kind);
      if (left > 0) {
        const q = next.players[seat]!;
        next = withPlayer(next, seat, { flags: { ...q.flags, [IMMUNITY_FLAG_PREFIX + kind]: left - 1 } });
      }
    }
  }
  const q = next.players[seat]!;
  return withPlayer(next, seat, { turn: { ...q.turn, statusGrantedThisTurn: false } });
}
```

```ts
/** skip 的封锁集（规格 §3.6 ②）。 */
export const SKIP_BLOCKED: ReadonlySet<string> = new Set(['MOVE', 'PLAY_CARD', 'ABILITY', 'ROLL_WHEEL', 'EQUIP_CARD', 'RECUPERATE']);
/** silence 的封锁集（§3.6 ③）——注意它是 skip 的真子集，故 skip 生效时 silence 被吸收。 */
export const SILENCE_BLOCKED: ReadonlySet<string> = new Set(['PLAY_CARD', 'ABILITY']);

/**
 * 某动作此刻是否被封。
 *
 * ⚠️ 只判**封锁集内**的动作类型；豁免族（结算期通道 / 对峙 / 认证 / 投票 / PASS）一律 false。
 *   写在这里的唯一理由是「三层同判」：枚举、引擎、UI 闸门必须读同一个函数，
 *   否则就是 B75.1 那种「判据顺序两处不一致」的重演。
 */
export function isActionBlocked(state: State, seat: number, type: string): boolean {
  if (statusOf(state, seat, 'skip_turn') > 0) { return SKIP_BLOCKED.has(type); }
  if (statusOf(state, seat, 'silence') > 0) { return SILENCE_BLOCKED.has(type); }
  return false;
}

/** 三层共用的**同一句**拒因（优先级：停手番 > 封口 > 力竭）。 */
export function blockedNotice(state: State, seat: number): string | null {
  if (statusOf(state, seat, 'skip_turn') > 0) { return '你被停了手番——这一回合只能结束行动。'; }
  if (statusOf(state, seat, 'silence') > 0) { return '你被封口了——本回合不能出牌或发动被动。'; }
  return null;
}
```

```js
// 试炼题面的唯一文本来源（规格 B77 §6.1）。
//
// 为什么要有这个文件：B22 时代 text 与 pred 成对编写、当时一致；B26 加难只换了 pred，
// 于是四套主题印出了 9–12 条与判定矛盾的题面，而 IR 只带 text（build-ir.mjs:138）⇒
// 玩家手里的牌写着条件 A、引擎判的是条件 B。卡牌早就由 renderCardText 派生，试炼是
// 全仓唯一"人抄的真相"，本文件把它并入同一条纪律。

import { renderEffect } from './theme-render.mjs';

/** 可派生判定白名单（§2.5：今天四套 48 条全覆盖 ⇒ 守卫钉住 fallback 数为 0）。 */
export const DERIVABLE_PREDICATES = new Set(['has_card_count', 'dice_is_at_least']);

export function isDerivable(pred) {
  return pred !== undefined && pred !== null && DERIVABLE_PREDICATES.has(pred.kind);
}

const TYPE_CN_FALLBACK = { motion: '移动', attack: '攻击', defense: '防御', utility: '辅助' };

/** 条件句：判定谓词 → 玩家看得懂的要求。 */
function renderCondition(pred, terms) {
  if (pred.kind === 'dice_is_at_least') { return `掷 ${String(pred.faces)} 面骰，点数 ≥ ${String(pred.amount)}`; }
  const f = pred.filter ?? {};
  const bits = [];
  if (f.color !== undefined) { bits.push(`${terms.colors[f.color] ?? (f.color === 'red' ? '红' : '黑')}色`); }
  if (f.type !== undefined) { bits.push(terms.types?.[f.type] ?? TYPE_CN_FALLBACK[f.type]); }
  const deck = terms.deck ?? '牌';
  return `展示 ${String(pred.count)} 张${bits.join('')}${deck}`;
}

/** 奖励句：直接复用卡牌的效果渲染器，两处口径同源。 */
function renderReward(then, terms) {
  return renderEffect(then, terms).replace(/[。.]$/, '');
}

/** 失败后果：主题包里 12 张全是 gate_fallback_pay，出现别的值就是数据变了。 */
function renderElse(elseEffect) {
  return elseEffect?.op === 'gate_fallback_pay' ? '未通过则本关改为付费' : `未通过则${renderEffect(elseEffect, {})}`;
}

/** 完整题面。白名单外的 pred 抛错——由调用方（生成器）转成 fallback 标记，绝不猜。 */
export function renderTrialText(effect, terms) {
  if (!isDerivable(effect?.pred)) { throw new Error('renderTrialText：判定不在可派生白名单内'); }
  return `${renderCondition(effect.pred, terms)}——通过后${renderReward(effect.then, terms)}（${renderElse(effect.else, terms)}）。`;
}
```

```js
export const EXPECTED_FALLBACK_COUNT = 0;   // B77：改这个数字 = 有人在绕过守卫，diff 里必须显眼

const TERM_LEAVES = [
  ['hp'], ['deck'], ['resource'], ['gate'],
  ['types', 'motion'], ['types', 'attack'], ['types', 'defense'], ['types', 'utility'],
  ['colors', 'red'], ['colors', 'black'],
];

function checkTrialTruth(theme, err) {
  for (const path of TERM_LEAVES) {
    const got = path.reduce((o, k) => (o === undefined || o === null ? undefined : o[k]), theme.terms);
    if (typeof got !== 'string' || got.length === 0) { err(`terms.${path.join('.')} 必须是非空字符串`); }
  }
  let fallback = 0;
  (theme.trials ?? []).forEach((t, i) => {
    const conditional = (t.effects ?? []).find((e) => e.op === 'conditional');
    const derivable = isDerivable(conditional?.pred);
    if (!derivable) { fallback += 1; if (typeof t.text !== 'string' || t.text.length === 0) { err(`trials[${i}] 非派生题面不得为空`); } return; }
    let want;
    try { want = renderTrialText(conditional, theme.terms); }
    catch (e) { err(`trials[${i}] 派生失败：${e.message}`); return; }
    if (t.text !== want) { err(`trials[${i}] 题面与判定不符：判[${want}] / 印[${t.text}]`); }
  });
  if (fallback !== EXPECTED_FALLBACK_COUNT) {
    err(`不可派生题面数 = ${String(fallback)}，守卫期望 ${String(EXPECTED_FALLBACK_COUNT)}：真要加非机械判定，改 EXPECTED_FALLBACK_COUNT 并在批次记录说明理由`);

}
```
