# B80 · 句子层三叶与身份称谓 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把引擎与界面里剩下 16 处主题词字面量（`rift`/`hazard`/`player`）与 8 处硬写身份名换成 `{rift}` `{hazard}` `{player}` `{identity:N}`，并让守卫族能同时看住「静态术语」与「运行时称谓」——**游戏行为一字不改**。

**Architecture:** 沿用 B79 的分工：`engine/**` 只发机制事实 + 占位符；取词发生在四个显示出口。本批新增**一类**占位符——`{identity:N}` 不是静态词表的叶子，而是"第 N 号座位在本套主题下的身份称谓"，由出口用 `state + theme` 解析（引擎不持有 theme，`State` 里只有 `identityId`，这是实测事实）。因此替换实现仍是两个函数、两种数据源：`expandThemeWords`（查 `terms`）与 `expandIdentityNames`（查座位）。

**Tech Stack:** TypeScript（`tsc --noEmit`，strict + `noUncheckedIndexedAccess` + `exactOptionalPropertyTypes`）、`node:test`、esbuild 单文件、13 道门（`tools/gate.mjs`）、无新增依赖。

**Spec:** `docs/superpowers/specs/2026-11-19-b80-句子层三叶与身份称谓-design.md`（r1.1）——计划按规格论证，执行时两份一起读。

```ts
/**
 * `{identity:N}` —— 第 N 号座位在本套主题下的身份称谓（B80 §2.3）。
 *
 * 与 `expandThemeWords` 分两个函数、两种数据源：那一个查**静态词表**（`terms`），
 * 这一个查**运行时主体**（座位 → 玩家身份 → 主题身份表）。合成一个"大替换"就把两件
 * 不同的事糊成一团，将来加 `{faction:N}` 时没有落点。
 *
 * ⚠️ 正则**每次调用新建**：带 `g` 的字面量正则带 `lastIndex` 状态，提到模块级常量复用
 *   会在第二次调用时从中途开始匹配（静默少替换，且只在同一条消息含两个占位符时暴露）。
 *   最后那条 `while` 循环不是多余的——`String.replace` 一次替换后若又冒出 `{identity:`，
 *   说明名字本身带占位符（数据被写坏），必须再来一轮直到干净。
 */
export function expandIdentityNames(raw: string, nameOf: (seat: number) => string): string {
  let out = raw;
  let guard = 0;
  while (/\{identity:/.test(out)) {
    out = out.replace(/\{identity:(\d+)\}/g, (_m, digits: string) => nameOf(Number(digits)));
    if ((/\{identity:\d+\}/.test(out)) && !/\{identity:\}/.test(out)) {
      guard += 1;
      if (guard > 4) { throw new Error(`身份称谓展开不收敛（句子：${raw}）`); }
      continue;
    }
    if (/\{identity\}|(?<!\d)\{identity:?(?!\d)/.test(out) && /\{identity:?\}/.test(out)) {
      throw new Error(`身份占位符格式非法：${out}`);
    }
    break;
  }
  const residue = /\{identity[^}]*\}*/.exec(out);
  if (residue !== null) { throw new Error(`身份占位符残留：${residue[0]}（句子：${raw}）`); }
  return out;
}
```

```ts
  // 1) 先拒畸形：出现 `{identity` 但不是 `{identity:<数字>}` 形状 ⇒ 直接抛错，不猜意图

  }
```

```js
/** 主题身份名（去重、排序）。从 theme.json 派生，**不手抄** —— 手抄的表在加第五套那天失效。 */
export function identityWordsFromThemes(root = 'themes') {
  const set = new Set();
  for (const entry of readdirSync(resolve(root))) {
    let ids;
    try { ids = JSON.parse(readFileSync(resolve(root, entry, 'theme.json'), 'utf8')).identities; } catch { continue; }
    for (const i of ids ?? []) { if (typeof i?.name === 'string' && i.name.length > 0) set.add(i.name); }
  }
  return [...set].sort((a, b) => a.localeCompare(b));
}

/** 主域禁字表 = 句子层叶词 ∪ 身份名（B80 R1/R1''）。 */
export function fullBanWords(root) {
  const ids = new Set(identityWordsFromThemes(root));
  return [...leafWordsFromThemes(root),
    ...[...ids].filter((w) => !leafWordsFromThemes(root).some((x) => x.word === w))
      .map((word) => ({ word, leaf: 'identity', themes: ['*'] }))];
```

```ts
// tests/gates/identity-token.test.ts —— B80 R9：身份占位符的可解析性（规格 §3 R9）
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { resolve } from 'node:path';

import { expandIdentityNames } from '../../engine/core/theme-text.ts';
import { installTheme } from '../../engine/core/setup.ts';
import { mk } from '../engine/fixtures.ts';
import { identityName } from '../../web/ui/terms.ts';

import type { ThemePack } from '../../engine/types/theme.ts';
import type { State } from '../../engine/core/state.ts';
import { stateFromJson } from '../../engine/core/serialize.ts';
import { setRecorder, type DecisionRecord } from '../sim/policy.mjs';

const themeOf = (id: string): ThemePack => JSON.parse(readFileSync(resolve('themes', id, 'theme.json'), 'utf8')) as ThemePack;
const IDS = ['xianxia2026', 'hybrid', 'scifi-ai', 'urban-psych'];

test('R9①：四套 × 24 局的每条 {identity:N} 日志都解析出本套身份名', async () => {
  let seen = 0;
  for (const id of IDS) {
    const theme = themeOf(id);
    const names = new Set(theme.identities.map((i) => i.name));
    for (const players of [3, 4, 5]) {
      for (let g = 0; g < 2; g += 1) {
        const seed = 20260 + g * 31 + players * 7;
        // ⚠️ `playOne` 返回的**不是终局 State**（它是仿真驱动；日志要从决策记录里取）
        //    ⇒ 照 B79 T14 的既有做法：setRecorder 收决策记录，取其中**日志最长**的那份状态。
        const bag: DecisionRecord[] = [];
        setRecorder(bag);
        try { await playOne(theme, players, seed); } finally { setRecorder(null); }
        assert.ok(bag.length > 0, `${id}/${String(players)} 人：一个决策点都没记录 ⇒ 真局没跑，前提不成立`);
        let state: State | null = null;
        for (const r of bag) {
          const s = stateFromJson(r.state);
          if (state === null || s.log.length > state.log.length) { state = s; }
        }
        assert.ok(state !== null, '前提：没取到任何真局状态');
        for (const entry of state.log) {
          if (!/\{identity:/.test(entry.message)) { continue; }
          const shown = expandIdentityNames(entry.message, (seat) => identityName(state, theme, seat));
          assert.equal(/\{/.test(shown), false, `${id}：展开后残留占位符 ${shown}`);
          const own = [...String(entry.message).matchAll(/\{identity:(\d+)\}/g)]
            .map((m) => identityName(state, theme, Number(m[1])));
          for (const n of own) {
            assert.ok(names.has(n), `${id}：解析出的称谓 ${n} 不在本套身份表里 ⇒ 词表与身份表漂移`);
          }
          seen += 1;
        }
      }
    }
  }
  assert.ok(seen > 0, '防空转：24 局里一条 {identity:N} 都没出现 ⇒ 接线根本没生效');
  console.log(`R9① 共核 ${String(seen)} 条身份称谓日志`);
});

});
```
