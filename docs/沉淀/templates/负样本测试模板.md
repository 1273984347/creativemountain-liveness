# 模板：常驻负样本测试

复制到 `tests/gates/<name>-negative.test.ts`，替换尖括号占位。

```ts
// 门禁第 <N> 道（<门名>）的**常驻负样本自证**。
//
// ⚠️ 必须用**隔离临时目录**：直接在共享工作树上改源码制造缺陷，
//   会让并行的 `node --test`（门 2）读到坏代码 ⇒ 门禁② 假红、无法区分真因。
//   ⇒ 复制到临时目录改**副本**，共享工作树全程只读。

import { test } from 'node:test';
import assert from 'node:assert/strict';
import { cpSync, existsSync, mkdtempSync, readFileSync, rmSync, writeFileSync } from 'node:fs';
import { spawnSync } from 'node:child_process';
import { tmpdir } from 'node:os';
import { resolve } from 'node:path';

const ROOT = process.cwd();

/** 建立隔离副本（只复制被测对象）。 */
function makeCopy(): string {
  const dir = mkdtempSync(resolve(tmpdir(), 'cm-neg-'));
  cpSync(resolve(ROOT, 'engine'), resolve(dir, 'engine'), { recursive: true });
  cpSync(resolve(ROOT, 'themes'), resolve(dir, 'themes'), { recursive: true });
  return dir;
}

/** 在副本里跑被测门禁（cwd 指向副本）。 */
function runGate(cwd: string): { status: number | null; out: string } {
  const r = spawnSync(process.execPath, [resolve(cwd, 'tests', '<entry>')], {
    cwd, encoding: 'utf8', shell: false, maxBuffer: 64 * 1024 * 1024,
  });
  return { status: r.status, out: (r.stdout ?? '') + (r.stderr ?? '') };
}

/** 破坏副本里的<机制>消费点。 */
function breakEngine(cwd: string): void {
  const p = resolve(cwd, 'engine', 'core', '<file>.ts');
  const src = readFileSync(p, 'utf8');
  const broken = src.replace(
    '<原文锚点>',
    '<破坏后的值>', // NEG-SAMPLE
  );
  assert.notEqual(broken, src, '负样本：应能在副本里破坏<机制>消费点');
  writeFileSync(p, broken, 'utf8');
}

test('负样本：破坏<机制> ⇒ 门 <N> 必红', () => {
  const dir = makeCopy();
  try {
    breakEngine(dir);
    const r = runGate(dir);
    assert.notEqual(r.status, 0, '破坏后门禁必须判红');
    assert.match(r.out, /<预期报错特征>/, '应报出<特征>');
  } finally {
    rmSync(dir, { recursive: true, force: true });
  }
});

test('正样本：未破坏 ⇒ 门 <N> 必绿（证明副本环境本身可用）', () => {
  const dir = makeCopy();
  try {
    const r = runGate(dir);
    assert.equal(r.status, 0, '未破坏时门禁必须绿：' + r.out.slice(-400));
  } finally {
    rmSync(dir, { recursive: true, force: true });
  }
});
```

## 交付前自检

1. 负样本必红 **且** 正样本必绿（两者都要有）。
2. `finally` 清理临时目录。
3. 跑完 `git status` 前后完全一致。
4. 若该脚本会在副本里运行 ⇒ 副本清单必须排除它自身（防无限套娃）。
5. 破坏点用**精确锚点**（`replace` 未命中时用 `assert.notEqual` 兜底，避免负样本静默失效）。
